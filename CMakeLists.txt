cmake_minimum_required(VERSION 3.20)
project(rudy_policyplane VERSION 0.1.0 LANGUAGES CXX)
option(POLICYPLANE_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /WX /permissive- /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Werror)
  if(POLICYPLANE_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()
add_library(policyplane_core OBJECT src/geometry.cpp)
target_sources(policyplane_core PRIVATE src/policy.cpp)
target_sources(policyplane_core PRIVATE src/engine.cpp)
target_sources(policyplane_core PRIVATE src/compare.cpp)
target_sources(policyplane_core PRIVATE src/report.cpp)
add_executable(policyplane src/main.cpp)
target_link_libraries(policyplane PRIVATE policyplane_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(policyplane_geometry tests/geometry.cpp)
  target_include_directories(policyplane_geometry PRIVATE src)
  target_link_libraries(policyplane_geometry PRIVATE policyplane_core)
  add_test(NAME geometry COMMAND policyplane_geometry)
endif()
if(BUILD_TESTING)
  find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
  add_test(NAME policy-oracle COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py" "$<TARGET_FILE:policyplane>" "${CMAKE_CURRENT_SOURCE_DIR}/examples")
endif()
